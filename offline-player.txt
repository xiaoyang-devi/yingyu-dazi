<script>
"use strict";
(() => {
  const data=JSON.parse(document.getElementById("cue-data").textContent), cues=data.cues;
  const byId=id=>document.getElementById(id), audio=byId("audio"), list=byId("rows"), feed=byId("feed");
  const follow=byId("follow"), progress=byId("status"), error=byId("error");
  const reduce=window.matchMedia("(prefers-reduced-motion: reduce)");
  const people=new Map(), buttons=[];
  const stamp=s=>Math.floor(s/60)+":"+String(Math.floor(s%60)).padStart(2,"0");
  const span=(cls,text)=>{const el=document.createElement("span");el.className=cls;el.textContent=text;return el;};
  byId("title").textContent=data.title; document.title=data.title;
  const fragment=document.createDocumentFragment();
  cues.forEach((cue,i)=>{
    if(!people.has(cue.speaker)) people.set(cue.speaker,people.size);
    const item=document.createElement("li");item.className=people.get(cue.speaker)%2?"row other":"row";
    const bubble=document.createElement("button");bubble.type="button";bubble.className="bubble";bubble.dataset.index=i;
    const who=span("who","");who.append(span("name",cue.speaker),span("mark","当前"),span("pos",stamp(cue.start)));
    const en=span("en",cue.en);en.lang="en";const zh=span("zh",cue.zh);zh.lang="zh-CN";
    bubble.append(who,en,zh);item.append(bubble);fragment.append(item);buttons.push(bubble);
  });
  list.append(fragment);
  let active=-1, frame=0, pending=null, previousLabel="", seenPlayback=false;
  function locate(t){
    let lo=0,hi=cues.length;
    while(lo<hi){const m=(lo+hi)>>>1;if(cues[m].start<=t)lo=m+1;else hi=m;}
    const prior=lo-1;return {prior,live:prior>=0 && t<cues[prior].end?prior:-1};
  }
  function center(i,smooth){
    if(i<0 || !follow.checked)return;
    const a=buttons[i].getBoundingClientRect(),f=feed.getBoundingClientRect();
    // A long utterance starts at its first line instead of hiding it above the viewport.
    const delta=a.height>f.height-18 ? a.top-f.top-9 : a.top-f.top+(a.height-f.height)/2;
    if(Math.abs(delta)>2)feed.scrollTo({top:feed.scrollTop+delta,behavior:smooth&&!reduce.matches?"smooth":"auto"});
  }
  function sync(force=false,smooth=false){
    const cursor=locate(audio.currentTime||0), live=audio.ended?-1:cursor.live;
    const changed=live!==active;
    if(changed){
      if(active>=0)buttons[active].removeAttribute("aria-current");
      active=live;if(active>=0)buttons[active].setAttribute("aria-current","true");
    }
    if(changed||force)center(live,smooth);
    let label;
    if(audio.ended)label="播放完成 · 共 "+cues.length+" 段";
    else if(live>=0)label=(audio.paused?"已暂停":"正在播放")+" · "+(live+1)+" / "+cues.length+" 段";
    else if(!seenPlayback && audio.currentTime===0)label="点播放或选一句 · 共 "+cues.length+" 段";
    else label=(audio.paused?"已暂停":"间奏停顿")+" · "+stamp(audio.currentTime||0)+" / "+stamp(data.duration);
    if(previousLabel!==label){progress.textContent=label;previousLabel=label;}
  }
  function tick(){
    sync(false,true);
    if(!audio.paused&&!audio.ended&&!document.hidden)frame=requestAnimationFrame(tick);
    else frame=0;
  }
  function restartClock(){
    if(frame)cancelAnimationFrame(frame);
    frame=0;sync(true,false);
    if(!audio.paused&&!audio.ended&&!document.hidden)frame=requestAnimationFrame(tick);
  }
  function playAt(t){
    pending=t;seenPlayback=true;
    if(audio.readyState>=1){audio.currentTime=t;pending=null;}
    const result=audio.play();
    if(result)result.catch(()=>{error.textContent="未能开始播放。请点上方播放器的播放按钮。";error.hidden=false;sync();});
    sync(true,false);
  }
  list.addEventListener("click",event=>{
    const target=event.target.closest("button[data-index]");
    if(target)playAt(cues[Number(target.dataset.index)].start);
  });
  byId("restart").addEventListener("click",()=>playAt(0));
  follow.addEventListener("change",()=>{if(follow.checked)sync(true,false);});
  audio.addEventListener("loadedmetadata",()=>{
    if(pending!==null){audio.currentTime=pending;pending=null;}
    sync(true,false);
  });
  audio.addEventListener("playing",()=>{seenPlayback=true;error.hidden=true;restartClock();});
  ["play","pause","ended","seeking","seeked","ratechange"].forEach(e=>audio.addEventListener(e,restartClock));
  audio.addEventListener("timeupdate",()=>sync(false,!audio.seeking&&!audio.paused));
  audio.addEventListener("error",()=>{error.textContent="这款浏览器未能解码内嵌的 WAV。请用支持 WAV 的浏览器打开此文件。";error.hidden=false;});
  document.addEventListener("visibilitychange",restartClock);
  window.addEventListener("pageshow",restartClock);
  window.addEventListener("resize",()=>sync(true,false));
  sync();
})();
</script>
